import re

def add_related_links(filepath, lang='zh'):
    with open(filepath, 'r', encoding='utf-8') as f:
        content = f.read()

    pattern = r'<li class="post-item">\s*<h2 class="post-title"><a href="([^"]+)">([^<]+)</a></h2>\s*<p class="post-meta">发布于 ([^<]+)</p>\s*</li>'
    posts = [(m.group(1), m.group(2)) for m in re.finditer(pattern, content)]

    # Build category pool per file
    categories = {'amazon-basics': [], 'wordpress': [], 'ollama-ai': [],
                  'vps-devops': [], 'docker': [], 'n8n': [], 'tool-review': []}
    for fname, title in posts:
        l = fname.lower()
        if 'amazon-basics' in l: categories['amazon-basics'].append((fname, title))
        elif 'wordpress' in l: categories['wordpress'].append((fname, title))
        elif any(x in l for x in ['ollama','lm-studio','jan','deepseek']): categories['ollama-ai'].append((fname, title))
        elif any(x in l for x in ['vps','digitalocean','cloud-server','vultr','doctl']): categories['vps-devops'].append((fname, title))
        elif 'docker' in l: categories['docker'].append((fname, title))
        elif 'n8n' in l: categories['n8n'].append((fname, title))
        else: categories['tool-review'].append((fname, title))

    def get_pool(fname):
        l = fname.lower()
        for kw, pool in categories.items():
            if kw in l: return pool
        return categories['tool-review']

    def replace_post(m):
        fname, title = m.group(1), m.group(2)
        pool = get_pool(fname)
        related = [(f, t) for f, t in pool if f != fname][:2]
        if not related: return m.group(0)
        related_title = 'Related Reading' if lang == 'en' else '相关阅读'
        html = f'\n            <div class="related-links">\n                <div class="related-links-title">{related_title}</div>\n'
        for rf, rt in related:
            dt = rt[:30]+'...' if len(rt) > 30 else rt
            html += f'                <a href="{rf}" class="related-link">{dt}</a>\n'
        html += '            </div>'
        return m.group(0).replace('        </li>', f'            {html}\n        </li>')

    new_content = re.sub(pattern, replace_post, content)
    with open(filepath, 'w', encoding='utf-8') as f:
        f.write(new_content)
    print(f'✅ {filepath}: {len(posts)} posts processed')

add_related_links('index.html', 'zh')
add_related_links('index-en.html', 'en')
PYEOF
